const { Client } = require("pg");
const fs = require("node:fs");

(async () => {
  const client = new Client({ connectionString: process.env.DATABASE_URL });
  let transactionStarted = false;

  try {
    await client.connect();
    await client.query("BEGIN");
    transactionStarted = true;

    await client.query(`
      DROP TABLE IF EXISTS solar_presence;
      DROP TABLE IF EXISTS solar_visits;
      DROP TABLE IF EXISTS solar_unique_presence;
      DROP TABLE IF EXISTS solar_unique_visits;
    `);

    await client.query(fs.readFileSync("database/analytics.sql", "utf8"));
    await client.query("COMMIT");
    transactionStarted = false;
    console.log("انجام شد؛ جدول‌های آمار Solar از نو ساخته شدند.");
  } catch (error) {
    if (transactionStarted) await client.query("ROLLBACK").catch(() => {});
    console.error("ریست ناموفق بود:", error.message);
    process.exitCode = 1;
  } finally {
    await client.end().catch(() => {});
  }
})();
